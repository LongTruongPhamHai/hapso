1.  BẮT ĐẦU
2.  
3.  // Giai đoạn 1: Kiểm tra tính hợp lệ của biên
4.  IF (Start hoặc Goal vi phạm vật cản/biên bản đồ) THEN:
5.      RETURN None
6.  ENDIF
7.  
8.  // Giai đoạn 2: Hoạch định hành lang hình học toàn cục (A*)
9.  P_initial ← EXECUTE_ASTAR(Start, Goal, GridMap)
10. IF (P_initial rỗng) THEN RETURN None
11.
12. // Giai đoạn 3: Tối ưu hóa bầy đàn thích nghi phân đoạn (PSO)
13. P_pso ← [P_initial[0]]
14. FOR i FROM 1 TO (Length(P_initial) – 2) DO:
15.     Prev ← P_pso[last]
16.     Next ← P_initial[i + 1]
17.
18.     // Khởi tạo quần thể khu trú
19.     Particles ← INITIALIZE_PARTICLES(Prev, Next, Omega)
20.
21.     FOR iteration FROM 1 TO Max_Iter DO:
22.         w ← CALCULATE_SIW(iteration)
23.         c1, c2 ← CALCULATE_TVAC(iteration)
24.
25.         FOR EACH particle IN Particles DO:
26.             UPDATE_VELOCITY(particle, w, c1, c2, G_best)
27.             UPDATE_POSITION(particle)
28.
29.             // Chiến lược học tập đối lập ngẫu nhiên
30.             Opp_Pos ← CALCULATE_SOBL(particle.pos)
31.             IF (Fitness(Opp_Pos) < Fitness(particle.pos)) THEN:
32.                 particle.pos ← Opp_Pos
33.             ENDIF
34.
35.             UPDATE_PERSONAL_BEST(particle)
36.         ENDFOR
37.         UPDATE_GLOBAL_BEST(G_best)
38.     ENDFOR
39.
40.     P_pso.APPEND(G_best.pos)
41. ENDFOR
42. P_pso.APPEND(Goal)
43.
44. // Giai đoạn 4: Xử lý làm mượt quỹ đạo
45. P_final ← BEZIER_SMOOTH(P_pso, GridMap)
46. RETURN P_final
47.
